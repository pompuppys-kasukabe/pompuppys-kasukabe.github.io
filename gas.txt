function getConfig() {
  const props = PropertiesService.getScriptProperties();

  return {
    // Notion設定
    NOTION_TOKEN: props.getProperty('NOTION_TOKEN'),

    // 写真データベース
    PHOTOS_DB_ID: props.getProperty('PHOTOS_DB_ID'),

    // メッセージデータベース
    MESSAGES_DB_ID: props.getProperty('MESSAGES_DB_ID'),

    // 活動カレンダーデータベース
    ACTIVITY_DB_ID: props.getProperty('ACTIVITY_DB_ID'),

    // Google Drive設定
    DRIVE_FOLDER_ID: props.getProperty('DRIVE_FOLDER_ID'),

    // 掲載枠とファイル名のマッピング
    SLOT_TO_FILENAME: {
      "Hero": "hero",
      "Gallery-1": "gallery_01",
      "Gallery-2": "gallery_02",
      "Gallery-3": "gallery_03",
      "Gallery-4": "gallery_04",
      "Gallery-5": "gallery_05",
      "Gallery-6": "gallery_06",
      "Gallery-7": "gallery_07",
      "Member-1": "member_01",
      "Member-2": "member_02",
      "Member-3": "member_03",
      "Member-4": "member_04",
      "Member-5": "member_05",
      "Member-6": "member_06",
      "Member-7": "member_07"
    }
  };
}

// ============================================
// Web App エンドポイント
// ============================================

function doGet(e) {
  const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : "getPhotos";

  try {
    if (action === "sync") {
      syncPhotosFromNotion();
      return jsonResponse({ success: true, message: "同期完了" });

    } else if (action === "getPhotos") {
      const data = buildPhotosData();
      return jsonResponse(data);

    } else if (action === "getMessages") {
      const messages = getMessages();
      return jsonResponse(messages);

    } else if (action === "getActivityData") {
      const activities = getActivityData();
      return jsonResponse(activities);

    } else {
      return jsonResponse({ error: "Unknown action" });
    }
  } catch (err) {
    console.error(err);
    return jsonResponse({ success: false, error: err.message });
  }
}

function jsonResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// ============================================
// 活動カレンダー取得
// ============================================

function getActivityData() {
  const CONFIG = getConfig();

  if (!CONFIG.NOTION_TOKEN || !CONFIG.ACTIVITY_DB_ID) {
    return {
      success: false,
      error: "Missing configuration (NOTION_TOKEN or ACTIVITY_DB_ID)"
    };
  }

  const url = `https://api.notion.com/v1/databases/${CONFIG.ACTIVITY_DB_ID}/query`;

  const options = {
    method: 'post',
    headers: {
      'Authorization': `Bearer ${CONFIG.NOTION_TOKEN}`,
      'Notion-Version': '2022-06-28',
      'Content-Type': 'application/json'
    },
    payload: JSON.stringify({
      filter: {
        property: 'Published',
        checkbox: {
          equals: true
        }
      },
      sorts: [
        {
          property: 'date',
          direction: 'descending'
        }
      ],
      page_size: 100
    }),
    muteHttpExceptions: true
  };

  try {
    const response = UrlFetchApp.fetch(url, options);
    const data = JSON.parse(response.getContentText());

    // エラーチェック
    if (data.error) {
      return {
        success: false,
        error: data.message || "Notion API error"
      };
    }

    // data.resultsが存在するかチェック
    if (!data.results) {
      return {
        success: false,
        error: "Unexpected API response structure"
      };
    }

    // データ整形
    const activities = data.results.map(page => {
      const props = page.properties;

      return {
        name: props["名前"]?.title?.[0]?.plain_text || "",
        date: props["date"]?.date?.start || null,
        type: props["type"]?.select?.name || "その他",
        typeColor: props["type"]?.select?.color || "default",
        startTime: props["Start Time"]?.rich_text?.[0]?.plain_text || "",
        endTime: props["End Time "]?.rich_text?.[0]?.plain_text || "",
        location: props["Location"]?.rich_text?.[0]?.plain_text || "",
        notes: props["Notes"]?.rich_text?.[0]?.plain_text || ""
      };
    });

    return {
      success: true,
      data: activities,
      count: activities.length
    };

  } catch (error) {
    return {
      success: false,
      error: error.toString()
    };
  }
}

// ============================================
// ファイルIDキャッシュ管理
// ============================================

function getFileIdCache() {
  const props = PropertiesService.getScriptProperties();
  const cache = props.getProperty("fileIdCache");
  return cache ? JSON.parse(cache) : {};
}

function setFileIdCache(cache) {
  const props = PropertiesService.getScriptProperties();
  props.setProperty("fileIdCache", JSON.stringify(cache));
}

// ============================================
// 写真データ取得(サイト表示用)
// ============================================

function buildPhotosData() {
  const CONFIG = getConfig();
  const pages = fetchNotionDatabase(CONFIG.PHOTOS_DB_ID, CONFIG.NOTION_TOKEN);
  const fileIdCache = getFileIdCache();

  const result = {
    updatedAt: new Date().toISOString().split("T")[0],
    hero: null,
    gallery: [],
    members: []
  };

  for (let i = 0; i < pages.length; i++) {
    const page = pages[i];
    const slot = getSlotFromPage(page);
    if (!slot || slot === "非公開") continue;

    const filename = CONFIG.SLOT_TO_FILENAME[slot];
    if (!filename) continue;

    const fileId = fileIdCache[filename];
    if (!fileId) continue;

    const item = {
      driveId: fileId,
      title: getTitleFromPage(page),
      alt: getDescriptionFromPage(page) || getTitleFromPage(page)
    };

    if (slot === "Hero") {
      result.hero = item;
    } else if (slot.indexOf("Gallery-") === 0) {
      const num = parseInt(slot.replace("Gallery-", ""));
      item.slot = num;
      item.size = getSizeFromPage(page);
      result.gallery.push(item);
    } else if (slot.indexOf("Member-") === 0) {
      const num = parseInt(slot.replace("Member-", ""));
      item.slot = num;
      item.name = "";
      result.members.push(item);
    }
  }

  result.gallery.sort((a, b) => a.slot - b.slot);
  result.members.sort((a, b) => a.slot - b.slot);

  return result;
}

// 応援メッセージ取得
// ============================================

function getMessages() {
  const CONFIG = getConfig();

  try {
    const url = `https://api.notion.com/v1/databases/${CONFIG.MESSAGES_DB_ID}/query`;

    const payload = {
      "filter": {
        "property": "承認済み",
        "checkbox": {
          "equals": true
        }
      },
      "sorts": [
        {
          "timestamp": "created_time",
          "direction": "descending"
        }
      ]
    };

    const options = {
      "method": "post",
      "headers": {
        "Authorization": `Bearer ${CONFIG.NOTION_TOKEN}`,
        "Notion-Version": "2022-06-28",
        "Content-Type": "application/json"
      },
      "payload": JSON.stringify(payload),
      "muteHttpExceptions": true
    };

    const response = UrlFetchApp.fetch(url, options);
    const data = JSON.parse(response.getContentText());

    if (!data.results) {
      Logger.log("Error: " + response.getContentText());
      return [];
    }

    const messages = data.results.map(page => {
      const props = page.properties;

      // Notionのcreated_timeから日付部分のみ抽出(YYYY-MM-DD)
      const createdTime = page.created_time || "";
      const dateOnly = createdTime ? createdTime.split('T')[0] : "";

      return {
        date: dateOnly,
        name: props["名前"] && props["名前"].rich_text[0] ? props["名前"].rich_text[0].plain_text : "匿名",
        message: props["メッセージ"] && props["メッセージ"].rich_text[0] ? props["メッセージ"].rich_text[0].plain_text : "",
        approved: props["承認済み"] ? props["承認済み"].checkbox : false
      };
    });

    return messages;

  } catch (e) {
    Logger.log("Error in getMessages: " + e.toString());
    return [];
  }
}

// ============================================
// Notion API(汎用)
// ============================================

function fetchNotionDatabase(dbId, token) {
  const url = `https://api.notion.com/v1/databases/${dbId}/query`;

  const options = {
    method: "post",
    headers: {
      "Authorization": `Bearer ${token}`,
      "Notion-Version": "2022-06-28",
      "Content-Type": "application/json"
    },
    payload: JSON.stringify({}),
    muteHttpExceptions: true
  };

  const response = UrlFetchApp.fetch(url, options);
  const data = JSON.parse(response.getContentText());

  if (data.error) {
    throw new Error("Notion API: " + data.message);
  }

  return data.results || [];
}

// ============================================
// ページプロパティ取得
// ============================================

function getSlotFromPage(page) {
  try {
    const prop = page.properties["掲載枠"];
    if (!prop || !prop.select) return null;
    return prop.select.name;
  } catch (e) {
    return null;
  }
}

function getTitleFromPage(page) {
  try {
    const prop = page.properties["写真タイトル"];
    if (!prop || !prop.title || !prop.title[0]) return "";
    return prop.title[0].plain_text;
  } catch (e) {
    return "";
  }
}

function getDescriptionFromPage(page) {
  try {
    const prop = page.properties["説明"];
    if (!prop || !prop.rich_text || !prop.rich_text[0]) return "";
    return prop.rich_text[0].plain_text;
  } catch (e) {
    return "";
  }
}

function getSizeFromPage(page) {
  try {
    const prop = page.properties["サイズ"];
    if (!prop || !prop.select) return "third";
    const sizeValue = prop.select.name || "";

    if (sizeValue.includes("Half") || sizeValue.includes("1/2")) return "half";
    if (sizeValue.includes("Third") || sizeValue.includes("1/3")) return "third";
    if (sizeValue.includes("Full") || sizeValue.includes("全幅")) return "full";
    if (sizeValue.includes("Large") || sizeValue.includes("大")) return "half";
    if (sizeValue.includes("Small") || sizeValue.includes("小")) return "third";

    return "third";
  } catch (e) {
    return "third";
  }
}

function getImageUrlFromPage(page) {
  try {
    const prop = page.properties["画像"];
    if (!prop || !prop.files || !prop.files[0]) return null;
    const file = prop.files[0];
    if (file.file && file.file.url) return file.file.url;
    if (file.external && file.external.url) return file.external.url;
    return null;
  } catch (e) {
    return null;
  }
}

// ============================================
// 画像同期(Notion → Google Drive)
// ============================================

function syncPhotosFromNotion() {
  const CONFIG = getConfig();
  console.log("=== 同期開始 ===");

  const pages = fetchNotionDatabase(CONFIG.PHOTOS_DB_ID, CONFIG.NOTION_TOKEN);
  console.log("取得ページ数: " + pages.length);

  const folder = DriveApp.getFolderById(CONFIG.DRIVE_FOLDER_ID);
  let fileIdCache = getFileIdCache();
  let syncCount = 0;

  for (let i = 0; i < pages.length; i++) {
    const page = pages[i];
    const slot = getSlotFromPage(page);
    if (!slot || slot === "非公開") continue;

    const filename = CONFIG.SLOT_TO_FILENAME[slot];
    if (!filename) continue;

    const imageUrl = getImageUrlFromPage(page);
    if (!imageUrl) {
      console.log("画像なし: " + slot);
      continue;
    }

    console.log("処理中: " + slot + " → " + filename + ".jpg");

    try {
      const blob = UrlFetchApp.fetch(imageUrl).getBlob();
      blob.setName(filename + ".jpg");

      const existingFiles = folder.getFilesByName(filename + ".jpg");
      while (existingFiles.hasNext()) {
        const oldFile = existingFiles.next();
        oldFile.setTrashed(true);
        console.log("既存ファイル削除: " + oldFile.getId());
      }

      const newFile = folder.createFile(blob);
      newFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      const fileId = newFile.getId();
      fileIdCache[filename] = fileId;

      console.log("完了: " + filename + ".jpg (ID: " + fileId + ")");
      syncCount++;

    } catch (err) {
      console.error("エラー: " + filename + " - " + err.message);
    }
  }

  setFileIdCache(fileIdCache);
  console.log("=== 同期完了: " + syncCount + "件 ===");

  return syncCount;
}

// ============================================
// トリガー管理
// ============================================

function createDailyTrigger() {
  const triggers = ScriptApp.getProjectTriggers();
  for (let i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === "syncPhotosFromNotion") {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }

  ScriptApp.newTrigger("syncPhotosFromNotion")
    .timeBased()
    .atHour(3)
    .everyDays(1)
    .create();

  console.log("トリガー設定完了: 毎日3時に自動実行");
}
