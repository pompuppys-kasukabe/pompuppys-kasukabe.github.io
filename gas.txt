      ).setMimeType(ContentService.MimeType.HTML);

    } else if (action === "getPhotos") {
      const data = buildPhotosData();
      return jsonResponse(data);

    } else if (action === "getMessages") {
      const messages = getMessages();
      return jsonResponse(messages);

    } else if (action === "getActivityData") {
      const activities = getActivityData();
      return jsonResponse(activities);

    } else if (action === "getInstagramFeed") {
      const feed = getInstagramFeed();
      return jsonResponse(feed);

    } else if (action === "getInstagramFeedFromNotion") {
      const feed = getInstagramFeedFromNotion();
      return jsonResponse(feed);

      const result = syncInstagramImageFileIds();

      return jsonResponse(getMessagesWithCount());

    } else {
      return jsonResponse({ error: "Unknown action" });

    }
  } catch (err) {
    console.error(err);
    return jsonResponse({ success: false, error: err.message });
  }
}

function jsonResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// ============================================
// 活動カレンダー取得
// ============================================

function getActivityData() {
  const CONFIG = getConfig();

  if (!CONFIG.NOTION_TOKEN || !CONFIG.ACTIVITY_DB_ID) {
    return {
      success: false,
      error: "Missing configuration (NOTION_TOKEN or ACTIVITY_DB_ID)"
    };
  }

  const url = `https://api.notion.com/v1/databases/${CONFIG.ACTIVITY_DB_ID}/query`;

  const options = {
    method: 'post',
    headers: {
      'Authorization': `Bearer ${CONFIG.NOTION_TOKEN}`,
      'Notion-Version': '2022-06-28',
      'Content-Type': 'application/json'
    },
    payload: JSON.stringify({
      // フィルターなし：Notionに追加された全ての予定を表示
      sorts: [
        {
          property: 'date',
          direction: 'descending'
        }
      ],
      page_size: 100
    }),
    muteHttpExceptions: true
  };

  try {
    const response = UrlFetchApp.fetch(url, options);
    const data = JSON.parse(response.getContentText());

    if (data.error) {
      return {
        success: false,
        error: data.message || "Notion API error"
      };
    }

    if (!data.results) {
      return {
        success: false,
        error: "Unexpected API response structure"
      };
    }

    const activities = data.results.map(page => {
      const props = page.properties;

      return {
        name: props["名前"]?.title?.[0]?.plain_text || "",
        date: props["date"]?.date?.start || null,
        type: props["type"]?.select?.name || "その他",
        typeColor: props["type"]?.select?.color || "default",
        startTime: props["Start Time"]?.rich_text?.[0]?.plain_text || "",
        endTime: props["End Time "]?.rich_text?.[0]?.plain_text || "",
        location: props["Location"]?.rich_text?.[0]?.plain_text || "",
        notes: props["Notes"]?.rich_text?.[0]?.plain_text || ""
      };
    });

    return {
      success: true,
      data: activities,
      count: activities.length
    };

  } catch (error) {
    return {
      success: false,
      error: error.toString()
    };
  }
}

// 写真データ取得(サイト表示用)
// ============================================

function buildPhotosData() {
  const CONFIG = getConfig();
  const pages = fetchNotionDatabase(CONFIG.PHOTOS_DB_ID, CONFIG.NOTION_TOKEN);
  const fileIdCache = getFileIdCache();

  const result = {
    updatedAt: new Date().toISOString().split("T")[0],
    hero: null,
    gallery: [],
    members: []
  };

  for (let i = 0; i < pages.length; i++) {
    const page = pages[i];
    const slot = getSlotFromPage(page);
    if (!slot || slot === "非公開") continue;

    const filename = CONFIG.SLOT_TO_FILENAME[slot];
    if (!filename) continue;

    const fileId = fileIdCache[filename];
    if (!fileId) continue;

    const item = {
      driveId: fileId,
      title: getTitleFromPage(page),
      alt: getDescriptionFromPage(page) || getTitleFromPage(page)
    };

    if (slot === "Hero") {
      result.hero = item;
    } else if (slot.indexOf("Gallery-") === 0) {
      const num = parseInt(slot.replace("Gallery-", ""));
      item.slot = num;
      item.size = getSizeFromPage(page);
      result.gallery.push(item);
    } else if (slot.indexOf("Member-") === 0) {
      const num = parseInt(slot.replace("Member-", ""));
      item.slot = num;
      item.name = "";
      result.members.push(item);
    }
  }

  result.gallery.sort((a, b) => a.slot - b.slot);
  result.members.sort((a, b) => a.slot - b.slot);

  return result;
}

// 応援メッセージ取得（キャッシュ対応・ページネーション対応版）
// ============================================

function getMessages() {
  const CONFIG = getConfig();

  // キャッシュをチェック（5分間有効）
  const cache = CacheService.getScriptCache();
  const cachedData = cache.get("messages_cache");

  if (cachedData) {
    console.log("メッセージ: キャッシュから取得");
    return JSON.parse(cachedData);
  }

  console.log("メッセージ: Notion APIから取得");

  try {
    const allMessages = [];
    let hasMore = true;
    let startCursor = undefined;

    // ページネーションで全件取得
    while (hasMore) {
      const url = `https://api.notion.com/v1/databases/${CONFIG.MESSAGES_DB_ID}/query`;

      const payload = {
        "filter": {
          "property": "承認済み",
          "checkbox": {
            "equals": true
          }
        },
        "sorts": [
          {
            "timestamp": "created_time",
            "direction": "descending"
          }
        ],
        "page_size": 100
      };

      if (startCursor) {
        payload.start_cursor = startCursor;
      }

      const options = {
        "method": "post",
        "headers": {
          "Authorization": `Bearer ${CONFIG.NOTION_TOKEN}`,
          "Notion-Version": "2022-06-28",
          "Content-Type": "application/json"
        },
        "payload": JSON.stringify(payload),
        "muteHttpExceptions": true
      };

      const response = UrlFetchApp.fetch(url, options);
      const data = JSON.parse(response.getContentText());

      if (!data.results) {
        Logger.log("Error: " + response.getContentText());
        break;
      }

      data.results.forEach(page => {
        const props = page.properties;
        const createdTime = page.created_time || "";
        const dateOnly = createdTime ? createdTime.split('T')[0] : "";

        // カテゴリ取得（セレクトプロパティ）
        let category = "一般";
        if (props["category"] && props["category"].select) {
          category = props["category"].select.name;
        } else if (props["カテゴリ"] && props["カテゴリ"].select) {
          category = props["カテゴリ"].select.name;
        }

        // ピックアップ取得（チェックボックス）
        let pickup = false;
        if (props["pickup"]) {
          pickup = props["pickup"].checkbox || false;
        } else if (props["ピックアップ"]) {
          pickup = props["ピックアップ"].checkbox || false;
        }

        allMessages.push({
          date: dateOnly,
          name: props["名前"] && props["名前"].rich_text[0] ? props["名前"].rich_text[0].plain_text : "匿名",
          message: props["メッセージ"] && props["メッセージ"].rich_text[0] ? props["メッセージ"].rich_text[0].plain_text : "",
          category: category,
          pickup: pickup,
          approved: props["承認済み"] ? props["承認済み"].checkbox : false
        });
      });

      hasMore = data.has_more;
      startCursor = data.next_cursor;

      console.log("取得済み: " + allMessages.length + "件, 続きあり: " + hasMore);
    }

    const result = { messages: allMessages };

    // キャッシュに保存（300秒 = 5分間）
    cache.put("messages_cache", JSON.stringify(result), 300);

    return result;

  } catch (e) {
    Logger.log("Error in getMessages: " + e.toString());
    return { messages: [] };
  }
}

// メッセージキャッシュをクリア（新しいメッセージを承認した後に実行）
function clearMessagesCache() {
  const cache = CacheService.getScriptCache();
  cache.remove("messages_cache");
  console.log("メッセージキャッシュをクリアしました");
}

function fetchNotionDatabase(dbId, token) {
  const url = `https://api.notion.com/v1/databases/${dbId}/query`;

  const options = {
    method: "post",
    headers: {
      "Authorization": `Bearer ${token}`,
      "Notion-Version": "2022-06-28",
      "Content-Type": "application/json"
    },
    payload: JSON.stringify({}),
    muteHttpExceptions: true
  };

  const response = UrlFetchApp.fetch(url, options);
  const data = JSON.parse(response.getContentText());

  if (data.error) {
    throw new Error("Notion API: " + data.message);
  }

  return data.results || [];
}

// ============================================
// ページプロパティ取得
// ============================================

function getSlotFromPage(page) {
  try {
    const prop = page.properties["掲載枠"];
    if (!prop || !prop.select) return null;
    return prop.select.name;
  } catch (e) {
    return null;
  }
}

function getTitleFromPage(page) {
  try {
    const prop = page.properties["写真タイトル"];
    if (!prop || !prop.title || !prop.title[0]) return "";
    return prop.title[0].plain_text;
  } catch (e) {
    return "";
  }
}

function getDescriptionFromPage(page) {
  try {
    const prop = page.properties["説明"];
    if (!prop || !prop.rich_text || !prop.rich_text[0]) return "";
    return prop.rich_text[0].plain_text;
  } catch (e) {
    return "";
  }
}

function getSizeFromPage(page) {
  try {
    const prop = page.properties["サイズ"];
    if (!prop || !prop.select) return "third";
    const sizeValue = prop.select.name || "";

    if (sizeValue.includes("Half") || sizeValue.includes("1/2")) return "half";
    if (sizeValue.includes("Third") || sizeValue.includes("1/3")) return "third";
    if (sizeValue.includes("Full") || sizeValue.includes("全幅")) return "full";
    if (sizeValue.includes("Large") || sizeValue.includes("大")) return "half";
    if (sizeValue.includes("Small") || sizeValue.includes("小")) return "third";

    return "third";
  } catch (e) {
    return "third";
  }
}

function getImageUrlFromPage(page) {
  try {
    const prop = page.properties["画像"];
    if (!prop || !prop.files || !prop.files[0]) return null;
    const file = prop.files[0];
    if (file.file && file.file.url) return file.file.url;
    if (file.external && file.external.url) return file.external.url;
    return null;
  } catch (e) {
    return null;
  }
}

// ============================================
// 画像同期(Notion → Google Drive)
// ============================================

function syncPhotosFromNotion() {
  const CONFIG = getConfig();
  console.log("=== 同期開始 ===");

  const pages = fetchNotionDatabase(CONFIG.PHOTOS_DB_ID, CONFIG.NOTION_TOKEN);
  console.log("取得ページ数: " + pages.length);

  const folder = DriveApp.getFolderById(CONFIG.DRIVE_FOLDER_ID);
  let fileIdCache = getFileIdCache();
  let syncCount = 0;

  for (let i = 0; i < pages.length; i++) {
    const page = pages[i];
    const slot = getSlotFromPage(page);
    if (!slot || slot === "非公開") continue;

    const filename = CONFIG.SLOT_TO_FILENAME[slot];
    if (!filename) continue;

    const imageUrl = getImageUrlFromPage(page);
    if (!imageUrl) {
      console.log("画像なし: " + slot);
      continue;
    }

    console.log("処理中: " + slot + " → " + filename + ".jpg");

    try {
      const blob = UrlFetchApp.fetch(imageUrl).getBlob();
      blob.setName(filename + ".jpg");

      const existingFiles = folder.getFilesByName(filename + ".jpg");
      while (existingFiles.hasNext()) {
        const oldFile = existingFiles.next();
        oldFile.setTrashed(true);
        console.log("既存ファイル削除: " + oldFile.getId());
      }

      const newFile = folder.createFile(blob);
      newFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      const fileId = newFile.getId();
      fileIdCache[filename] = fileId;

      console.log("完了: " + filename + ".jpg (ID: " + fileId + ")");
      syncCount++;

    } catch (err) {
      console.error("エラー: " + filename + " - " + err.message);
    }
  }

  setFileIdCache(fileIdCache);
  console.log("=== 同期完了: " + syncCount + "件 ===");

  return syncCount;
}

function createDailyTrigger() {
  const triggers = ScriptApp.getProjectTriggers();
  for (let i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === "syncPhotosFromNotion") {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }

  ScriptApp.newTrigger("syncPhotosFromNotion")
    .timeBased()
    .atHour(3)
    .everyDays(1)
    .create();

  console.log("トリガー設定完了: 毎日3時に自動実行");
}

function getInstagramFeed() {
  const CONFIG = getConfig();

  if (!CONFIG.INSTAGRAM_ACCESS_TOKEN) {
    return {
      success: false,
      error: "Missing INSTAGRAM_ACCESS_TOKEN"
    };
  }

  const fields = 'id,caption,media_type,media_url,permalink,thumbnail_url,timestamp';
  const url = `https://graph.instagram.com/me/media?fields=${fields}&access_token=${CONFIG.INSTAGRAM_ACCESS_TOKEN}&limit=12`;

  try {
    const response = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    const data = JSON.parse(response.getContentText());

    if (data.error) {
      return {
        success: false,
        error: data.error.message || "Instagram API error"
      };
    }

    if (!data.data || data.data.length === 0) {
      return {
        success: true,
        data: [],
        count: 0
      };
    }

    const posts = data.data.map(function(post) {
      return {
        id: post.id,
        caption: post.caption || "",
        mediaType: post.media_type,
        mediaUrl: post.media_type === 'VIDEO' ? (post.thumbnail_url || post.media_url) : post.media_url,
        permalink: post.permalink,
        timestamp: post.timestamp
      };
    }).filter(function(post) {
      return post.mediaType === 'IMAGE' || post.mediaType === 'CAROUSEL_ALBUM';
    });

    return {
      success: true,
      data: posts.slice(0, 6),
      count: posts.length
    };

  } catch (error) {
    return {
      success: false,
      error: error.toString()
    };
  }
}

// Instagramトークンリフレッシュ
// ============================================

function refreshInstagramToken() {
  const CONFIG = getConfig();

  if (!CONFIG.INSTAGRAM_ACCESS_TOKEN) {
    console.log("No Instagram access token found");
    return;
  }

  const url = `https://graph.instagram.com/refresh_access_token?grant_type=ig_refresh_token&access_token=${CONFIG.INSTAGRAM_ACCESS_TOKEN}`;

  try {
    const response = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    const data = JSON.parse(response.getContentText());

    if (data.error) {
      console.error("Token refresh error: " + data.error.message);
      return;
    }

    PropertiesService.getScriptProperties().setProperty('INSTAGRAM_ACCESS_TOKEN', data.access_token);

    console.log("Instagram token refreshed successfully. Expires in: " + data.expires_in + " seconds");

  } catch (error) {
    console.error("Error refreshing Instagram token: " + error.toString());
  }
}

// Instagram画像ファイルIDを更新（PNG専用・高速版）
function syncInstagramImageFileIds() {
  const CONFIG = getConfig();
  const fileIds = {};

  try {
    const folder = DriveApp.getFolderById(CONFIG.DRIVE_FOLDER_ID);

    for (let i = 1; i <= 6; i++) {
      const fileName = `insta${i}.png`;
      const files = folder.getFilesByName(fileName);

      if (files.hasNext()) {
        fileIds[`insta${i}`] = files.next().getId();
        console.log(`Found: ${fileName}`);
      } else {
        console.log(`Not found: ${fileName}`);
      }
    }

    setInstagramFileIdCache(fileIds);
    console.log("Instagram画像キャッシュ更新完了:", JSON.stringify(fileIds));
    return fileIds;

  } catch (error) {
    console.error("Instagram画像キャッシュ更新エラー:", error);
    return {};
  }
}

// Instagram投稿取得（Notion連携版）- 高速版
// ============================================

function getInstagramFeedFromNotion() {
  const CONFIG = getConfig();

  if (!CONFIG.NOTION_TOKEN) {
    return { success: false, error: "Missing NOTION_TOKEN" };
  }

  if (!CONFIG.NOTION_INSTAGRAM_DB_ID) {
    return { success: false, error: "Missing NOTION_INSTAGRAM_DB_ID" };
  }

  try {
    // キャッシュからファイルIDを取得（高速）
    let imageFileIds = getInstagramFileIdCache();

    // キャッシュが空なら同期（初回のみ）
    if (Object.keys(imageFileIds).length === 0) {
      console.log("キャッシュが空のため同期を実行");
      imageFileIds = syncInstagramImageFileIds();
    }

    // Notionデータベースをクエリ
    const url = `https://api.notion.com/v1/databases/${CONFIG.NOTION_INSTAGRAM_DB_ID}/query`;

    const options = {
      method: 'post',
      headers: {
        'Authorization': `Bearer ${CONFIG.NOTION_TOKEN}`,
        'Notion-Version': '2022-06-28',
        'Content-Type': 'application/json'
      },
      payload: JSON.stringify({
        sorts: [{ property: '表示順', direction: 'ascending' }],
        filter: {
          and: [
            { property: '表示順', number: { is_not_empty: true } },
            { property: '表示順', number: { less_than_or_equal_to: 6 } }
          ]
        },
        page_size: 6
      }),
      muteHttpExceptions: true
    };

    const response = UrlFetchApp.fetch(url, options);
    const data = JSON.parse(response.getContentText());

    if (data.object === 'error') {
      return { success: false, error: `Notion API error: ${data.message}` };
    }

    if (!data.results || data.results.length === 0) {
      return { success: true, data: [], count: 0 };
    }

    // データ整形
    const posts = data.results.map(function(page) {
      const properties = page.properties;

      let instagramUrl = '';
      if (properties['Instagram URL'] && properties['Instagram URL'].url) {
        instagramUrl = properties['Instagram URL'].url;
      }

      let caption = '';
      if (properties['キャプション'] && properties['キャプション'].rich_text && properties['キャプション'].rich_text.length > 0) {
        caption = properties['キャプション'].rich_text[0].plain_text;
      }

      let displayOrder = 1;
      if (properties['表示順'] && properties['表示順'].number) {
        displayOrder = properties['表示順'].number;
      }

      // キャッシュからファイルIDを取得（高速）
      const driveId = imageFileIds[`insta${displayOrder}`] || '';

      return {
        id: `post${displayOrder}`,
        url: instagramUrl,
        driveId: driveId,
        caption: caption,
        displayOrder: displayOrder
      };
    }).filter(post => post.url && post.driveId);

    return {
      success: true,
      data: posts,
      count: posts.length
    };

  } catch (error) {
    return { success: false, error: error.toString() };
  }
}

// ============================================
// 応援メッセージ（千羽鶴ページ用・カウント付き・ページネーション対応）
// ============================================

function getMessagesWithCount() {
  const CONFIG = getConfig();

  // キャッシュをチェック（5分間有効）
  const cache = CacheService.getScriptCache();
  const cachedData = cache.get("messages_with_count_cache");

  if (cachedData) {
    console.log("メッセージ（カウント付き）: キャッシュから取得");
    return JSON.parse(cachedData);
  }

  console.log("メッセージ（カウント付き）: Notion APIから取得");

  try {
    const allMessages = [];
    const pickupMessages = [];
    let hasMore = true;
    let startCursor = undefined;

    // ページネーションで全件取得
    while (hasMore) {
      const url = `https://api.notion.com/v1/databases/${CONFIG.MESSAGES_DB_ID}/query`;

      const payload = {
        "filter": {
          "property": "承認済み",
          "checkbox": {
            "equals": true
          }
        },
        "sorts": [
          {
            "timestamp": "created_time",
            "direction": "descending"
          }
        ],
        "page_size": 100
      };

      if (startCursor) {
        payload.start_cursor = startCursor;
      }

      const options = {
        "method": "post",
        "headers": {
          "Authorization": `Bearer ${CONFIG.NOTION_TOKEN}`,
          "Notion-Version": "2022-06-28",
          "Content-Type": "application/json"
        },
        "payload": JSON.stringify(payload),
        "muteHttpExceptions": true
      };

      const response = UrlFetchApp.fetch(url, options);
      const data = JSON.parse(response.getContentText());

      if (!data.results) {
        break;
      }

      data.results.forEach((page) => {
        const props = page.properties;
        const createdTime = page.created_time || "";
        const dateOnly = createdTime ? createdTime.split('T')[0] : "";

        const message = {
          id: page.id,
          index: allMessages.length,
          date: dateOnly,
          name: props["名前"]?.rich_text?.[0]?.plain_text || "匿名",
          message: props["メッセージ"]?.rich_text?.[0]?.plain_text || "",
          pickup: props["pickup"]?.checkbox || false,
          category: props["category"]?.select?.name || "一般"
        };

        allMessages.push(message);

        if (message.pickup) {
          pickupMessages.push(message);
        }
      });

      hasMore = data.has_more;
      startCursor = data.next_cursor;

      console.log("取得済み: " + allMessages.length + "件, 続きあり: " + hasMore);
    }

    const result = {
      success: true,
      messages: allMessages,
      pickupMessages: pickupMessages,
      count: allMessages.length,
      goal: 1000,
      percentage: Math.min(Math.round((allMessages.length / 1000) * 100), 100)
    };

    // キャッシュに保存（300秒 = 5分間）
    cache.put("messages_with_count_cache", JSON.stringify(result), 300);

    return result;

  } catch (e) {
    console.error("Error in getMessagesWithCount: " + e.toString());
    return { success: false, messages: [], count: 0, pickupMessages: [], error: e.toString() };
  }
}

// 千羽鶴キャッシュをクリア
function clearMessagesWithCountCache() {
  const cache = CacheService.getScriptCache();
  cache.remove("messages_with_count_cache");
  console.log("千羽鶴キャッシュをクリアしました");
}

// 両方のキャッシュをクリア（新しいメッセージ承認後に実行）
function clearAllMessagesCache() {
  clearMessagesCache();
  clearMessagesWithCountCache();
  console.log("全てのメッセージキャッシュをクリアしました");
}

  var result = getMessagesWithCount();
  Logger.log("総数: " + result.count + " / 目標: " + result.goal);
  Logger.log("達成率: " + result.percentage + "%");
  Logger.log("ピックアップ: " + result.pickupMessages.length + "件");
  Logger.log(JSON.stringify(result, null, 2));
}
